#include <iostream>
using namespace std;

int main() {
    int a[5];
    int *p = a;

    cout << "Enter 5 numbers: ";
    for (int i = 0; i < 5; i++) {
        cin >> *(p + i);
    }

    int largest = *p;

    for (int i = 1; i < 5; i++) {
        if (*(p + i) > largest) {
            largest = *(p + i);
        }
    }

    cout << "Largest = " << largest;

    return 0;
}
